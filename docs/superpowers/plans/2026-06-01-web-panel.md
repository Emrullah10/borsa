# Web Panel (Faz 1E) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scalp bot için koyu temalı, 3 panelli bir web izleme paneli kur; ilk aşamada mock data ile çalışsın.

**Architecture:** Vite + React tek sayfa uygulama. Veri katmanı tek bir `mockData.js` modülünden gelir ve `signals` DB şemasının camelCase karşılığıyla şekillenir (ileride canlı API'ye geçiş için drop-in). Zustand global UI state (seçili sinyal) tutar. Layout 3 sütun: sinyal listesi (AG Grid) | fiyat grafiği + indikatör grafiği (ECharts) | açık sinyaller (AG Grid). Üst bar canlı fiyat + servis durumu (mock).

**Tech Stack:** Vite, React 18, MUI v5 (koyu tema), AG Grid Community, ECharts (echarts-for-react), Zustand. Vitest + @testing-library/react test için.

**Dizin:** `borsa/frontend/` (ayrı npm projesi, backend workspace'inden bağımsız).

**Renk konvansiyonu:** Arka plan `#0d1117`, LONG = yeşil `#26a69a`, SHORT = kırmızı `#ef5350`, accent metin `#c9d1d9`.

- [ ] **Step 3: mockData.js yaz**

```js
const SYMBOLS = ['BTCUSDT', 'ETHUSDT'];
const BASE = { BTCUSDT: 78000, ETHUSDT: 3100 };

function gen(seed) {
  // deterministik pseudo-random
  let x = Math.sin(seed) * 10000;
  return x - Math.floor(x);
}

export const mockSignals = Array.from({ length: 60 }, (_, i) => {
  const symbol = SYMBOLS[i % 2];
  const direction = gen(i) > 0.5 ? 'long' : 'short';
  const entry = BASE[symbol] * (1 + (gen(i + 100) - 0.5) * 0.01);
  const dist = entry * 0.0008;
  const stop = direction === 'long' ? entry - dist : entry + dist;
  const target = direction === 'long' ? entry + dist * 1.5 : entry - dist * 1.5;
  return {
    id: `sig-${i}`,
    symbol,
    direction,
    entryPrice: parseFloat(entry.toFixed(2)),
    stopPrice: parseFloat(stop.toFixed(2)),
    targetPrice: parseFloat(target.toFixed(2)),
    rrRatio: 1.5,
    confluenceScore: parseFloat((0.65 + gen(i + 7) * 0.3).toFixed(3)),
    createdAt: new Date(Date.now() - i * 5 * 60 * 1000).toISOString(),
    status: i < 6 ? 'open' : 'closed',
  };
});

export const mockCandles = Array.from({ length: 120 }, (_, i) => {
  const base = BASE.BTCUSDT;
  const open = base * (1 + (gen(i) - 0.5) * 0.004);
  const close = open * (1 + (gen(i + 50) - 0.5) * 0.003);
  const high = Math.max(open, close) * (1 + gen(i + 9) * 0.001);
  const low = Math.min(open, close) * (1 - gen(i + 3) * 0.001);
  return {
    timestamp: Date.now() - (120 - i) * 60 * 1000,
    open: parseFloat(open.toFixed(2)),
    high: parseFloat(high.toFixed(2)),
    low: parseFloat(low.toFixed(2)),
    close: parseFloat(close.toFixed(2)),
    volume: parseFloat((gen(i + 11) * 500).toFixed(2)),
  };
});

export const mockServiceStatus = {
  marketData: { up: true, port: 3101 },
  signalEngine: { up: true, port: 3102 },
  livePrice: { BTCUSDT: BASE.BTCUSDT, ETHUSDT: BASE.ETHUSDT },
};
```

import ReactECharts from 'echarts-for-react';
import { Box } from '@mui/material';
import { mockCandles } from '../data/mockData.js';
import { COLORS } from '../theme.js';

export function buildChartOption(candles) {
  const times = candles.map((c) => new Date(c.timestamp).toLocaleTimeString('tr-TR', { hour: '2-digit', minute: '2-digit' }));
  const ohlc = candles.map((c) => [c.open, c.close, c.low, c.high]);
  const vol = candles.map((c) => c.volume);
  return {
    backgroundColor: 'transparent',
    grid: [
      { left: 50, right: 20, top: 20, height: '60%' },
      { left: 50, right: 20, top: '78%', height: '16%' },
    ],
    xAxis: [
      { type: 'category', data: times, gridIndex: 0 },
      { type: 'category', data: times, gridIndex: 1, axisLabel: { show: false } },
    ],
    yAxis: [
      { scale: true, gridIndex: 0 },
      { gridIndex: 1, axisLabel: { show: false } },
    ],
    series: [
      {
        type: 'candlestick', data: ohlc, xAxisIndex: 0, yAxisIndex: 0,
        itemStyle: { color: COLORS.long, color0: COLORS.short, borderColor: COLORS.long, borderColor0: COLORS.short },
      },
      { type: 'bar', data: vol, xAxisIndex: 1, yAxisIndex: 1, itemStyle: { color: '#444c56' } },
    ],
  };
}

export default function PriceChart() {
  return (
    <Box sx={{ height: '100%', minHeight: 300 }}>
      <ReactECharts option={buildChartOption(mockCandles)} style={{ height: '100%' }} />
    </Box>
  );
}
